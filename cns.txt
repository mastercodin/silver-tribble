string mod2div(string dividend, string divisor) {
int pick = divisor.length();
string tmp = dividend.substr(0, pick);
int n = dividend.length();
while (pick < n) {
if (tmp[0] == '1')
tmp = xor1(divisor, tmp) + dividend[pick];
else
tmp = xor1(string(pick, '0'), tmp) + dividend[pick];
pick++;
}

#define MAX_SEQ 7
class GoBackN {
int base = 0;
int nextSeqNum = 0;
int windowSize;
vector<bool> acked;
vector<bool> initiallyLost;
public:
GoBackN(int w, const vector<bool> &frameLost) : windowSize(w),
acked(MAX_SEQ, false), initiallyLost(frameLost) {}
void sender() {
cout << "\nGo-Back-N Sender:\n";
while (base < MAX_SEQ) {
while (nextSeqNum < base + windowSize && nextSeqNum < MAX_SEQ) {
cout << "Sending frame " << nextSeqNum << endl;
// On first send, check initial loss
if (initiallyLost[nextSeqNum]) {
cout << "Frame " << nextSeqNum << " lost." << endl;
initiallyLost[nextSeqNum] = false; // Clear loss on
retransmission
break;
} else {
cout << "ACK received for frame " << nextSeqNum << endl;
acked[nextSeqNum] = true;
}
nextSeqNum++;
}
while (base < MAX_SEQ && acked[base]) {
base++;
}
if (base < MAX_SEQ && !acked[base]) {
cout << "Timeout or Loss detected. Retransmitting from frame
" << base << endl;
nextSeqNum = base;
}
}
cout << "All frames sent and acknowledged in Go-Back-N\n";
}
};
class SelectiveRepeat {
int base = 0;
int nextSeqNum = 0;
int windowSize;
vector<bool> acked;
vector<bool> initiallyLost;
public:
SelectiveRepeat(int w, const vector<bool> &frameLost) : windowSize(w),
acked(MAX_SEQ, false), initiallyLost(frameLost) {}
void sender() {
cout << "\nSelective Repeat Sender:\n";
while (base < MAX_SEQ) {
while (nextSeqNum < base + windowSize && nextSeqNum < MAX_SEQ) {
cout << "Sending frame " << nextSeqNum << endl;
if (initiallyLost[nextSeqNum]) {
cout << "Frame " << nextSeqNum << " lost." << endl;
initiallyLost[nextSeqNum] = false; // Clear loss on
retransmission
} else {
cout << "ACK received for frame " << nextSeqNum << endl;
acked[nextSeqNum] = true;
}
nextSeqNum++;
}
while (base < MAX_SEQ && acked[base]) {
base++;
}
for (int i = base; i < base + windowSize && i < MAX_SEQ; i++) {
if (!acked[i]) {
cout << "Retransmitting frame " << i << endl;
cout << "ACK received for frame " << i << " after
retransmission\n";
acked[i] = true;
}
}
}
cout << "All frames sent and acknowledged in Selective Repeat\n";
}
};
